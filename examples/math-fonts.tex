\begin{frame}
  \frametitle{Completing the Square}

  \large

  First, complete the square
  
  \begin{align}
    ax^{2} + bx +c& =0 \\ \pause
    x^{2} + \frac{b}{a}x + \frac{c}{a} & = 0 \\ \pause
    x^{2} + \frac{b}{a}x + \bigl(\frac{b}{2a}\bigr)^{2} 
      & = -\frac{c}{a} + \bigl(\frac{b}{2a}\bigr)^{2} \\ \pause
    \bigl(x + \frac{b}{2a}\bigr)^{2} 
      & = -\frac{c}{a} + \bigl(\frac{b}{2a}\bigr)^{2}
  \end{align}
\end{frame}

\begin{frame}
 \frametitle{Solve for~$x$}
 \large

 Now, solve for~$x$
 
  \begin{align}
    \bigl(x + \frac{b}{2a}\bigr)^{2}
      &= -\frac{c}{a} + \bigl(\frac{b}{2a}\bigr)^{2} \\ \pause
    x + \bigl(\frac{b}{2a}\bigr)
      &= \pm \sqrt{-\frac{c}{a} + \bigl(\frac{b}{2a}\bigr)^{2}} \\ \pause
    x &= -\bigl(\frac{b}{2a}\bigr)
      \pm \sqrt{-\frac{c}{a} + \bigl(\frac{b}{2a}\bigr)^{2}} \\ \pause
     x &= \frac{-b \pm \sqrt{b^{2} - 4ac}}{2a}
  \end{align}
    
\end{frame}

\begin{frame}
 \frametitle{The Quadratic Formula}
 \large

 The solution of the quadratic equation.

 \[
   ax^{2} + bx + c = 0
 \]

 where~$a$ is non-zero, is given by the formula

  \[
    x = \frac{-b \pm \sqrt{b^{2} - 4ac}}{2a}
  \]

\end{frame}
